def load_json(path):
    with path.open(
        encoding="utf-8"
    ) as f:
        return json.load(f)


def sha256(path):
    h = hashlib.sha256()

    with path.open("rb") as f:
        for block in iter(
            lambda: f.read(1024 * 1024),
            b""
        ):
            h.update(block)

    return h.hexdigest()

    interpretation=""

            "PASS"

            observed,

        "expected":
            expected,
